# โหมด 3 มิติด้วย three.js — แผนลงมือ

> **สำหรับผู้ลงมือ:** ทำทีละงานตามลำดับ ทุกงานจบด้วยของที่เปิดดูได้และ commit ได้
> ช่องติ๊ก `- [ ]` ใช้ไล่ความคืบหน้า

**เป้าหมาย:** เพิ่มตัววาดโหมด 3 มิติตัวที่สองด้วย three.js ที่วาดก้อนควันแบบ raymarch
จนภูเขาบังควันได้จริง โดยตัวเดิม (MapLibre) ยังใช้งานได้ครบเป็น legacy

**สถาปัตยกรรม:** ฉากท้องถิ่นหน่วยเมตร **แกน Z ชี้ขึ้น** พิกัด `x, y` ในฉากคือพิกัดเอนจินตรงๆ
ภูมิประเทศเป็น `PlaneGeometry` ที่ยกจาก DEM ก้อนควันเป็นกล่องเดียวที่ fragment shader
เดินรังสีผ่าน `Data3DTexture` และหยุดเมื่อจมใต้ผิวดินที่อ่านจาก texture ความสูง

**เทคโนโลยี:** three.js 0.186.0 · WebGL2 (`Data3DTexture` + `sampler3D` ต้องการ) · vitest

**สเปก:** `docs/superpowers/specs/2026-09-14-three-3d-design.md` — อ่านคู่กันเสมอ

```js
// app/src/map3d/sky-palette.js
/**
 * sky-palette — ฟ้าและดวงอาทิตย์ตามชั่วโมง ใช้ร่วมกันระหว่างตัววาดสองตัว
 *
 * `skyFor` ย้ายมาจาก `app.js` ทั้งดุ้น **ไม่แก้ค่าสีแม้แต่ตัวเดียว** เพราะถ้าสองโหมด
 * ใช้จานสีคนละชุด คนที่สลับไปมาจะเห็นเวลาในวันไม่ตรงกัน แล้วจะไม่มีใครรู้ว่าอันไหนถูก
 *
 * `sunDirection` เป็นของใหม่ — MapLibre ใช้ hillshade ที่ล็อกมุมแสงไว้ที่ 315°
 * ส่วน three.js ใช้แสงจริงส่องเมช จึงต้องรู้ว่าดวงอาทิตย์อยู่ตรงไหนจริงๆ
 */

const HOUR_RE = /T(\d{2})/;
function hourOf(hourKey) {
  const m = HOUR_RE.exec(hourKey || '');
  const h = m ? +m[1] : NaN;
  return Number.isFinite(h) ? h : 12;      // คีย์เสีย → เที่ยงวัน เหมือนพฤติกรรมเดิมใน app.js
}

/** จานสีฟ้า/ขอบฟ้า/หมอก — ย้ายมาจาก app.js ห้ามแก้ค่า */
export function skyFor(hourKey) {
  const hh = hourOf(hourKey);
  if (hh < 6 || hh >= 19) return { sky:'#0b1220', hor:'#1d2a3d', fog:'#141d2a' };
  if (hh < 8)  return { sky:'#4a5f86', hor:'#e0a765', fog:'#c8b49a' };
  if (hh < 16) return { sky:'#5f8fc4', hor:'#b9cbdc', fog:'#c3ceda' };
  return { sky:'#3f5c88', hor:'#e09a5e', fog:'#c2ae97' };
}

export function isDaylight(hourKey) { const h = hourOf(hourKey); return h >= 6 && h < 19; }

/**
 * ทิศดวงอาทิตย์ในระบบพิกัดฉาก (x ตะวันออก · y เหนือ · z ขึ้น)
 *
 * ใช้สูตรตำแหน่งดวงอาทิตย์แบบย่อ คลาดเคลื่อนระดับองศา ซึ่งพอสำหรับ**ทิศเงาเขา**
 * ไม่ได้เอาไปคำนวณรังสีหรือค่าใดๆ ที่รายงานเป็นตัวเลข จึงไม่ต้องใช้สูตรเต็ม
 * วันที่ตรึงกลางเดือนมีนาคมเมื่อคีย์ไม่มีวันที่ เพราะฤดูเผาของภาคเหนืออยู่แถวนั้น
 */
export function sunDirection(hourKey, lat) {
  const h = hourOf(hourKey);
  const d = new Date((hourKey || '').slice(0, 10) || '2026-03-15');
  const doy = Number.isFinite(d.getTime())
    ? Math.floor((d - new Date(d.getFullYear(), 0, 0)) / 86400000) : 74;
  const rad = Math.PI / 180;
  const decl = 23.44 * rad * Math.sin(2 * Math.PI * (284 + doy) / 365);
  const ha = (h - 12) * 15 * rad;          // มุมชั่วโมง เที่ยง = 0 เช้าเป็นลบ
  const la = lat * rad;
  const alt = Math.asin(Math.sin(la) * Math.sin(decl) + Math.cos(la) * Math.cos(decl) * Math.cos(ha));
  // อะซิมุทวัดจากทิศเหนือตามเข็มนาฬิกา
  const az = Math.atan2(-Math.sin(ha), Math.tan(decl) * Math.cos(la) - Math.sin(la) * Math.cos(ha));
  return { x: Math.sin(az) * Math.cos(alt), y: Math.cos(az) * Math.cos(alt), z: Math.sin(alt) };
}
```

```js
 * basemap — โมเสกภาพดาวเทียมสำหรับปูผิวภูมิประเทศในฉาก three.js
 *

 * จึงให้งบ 6×6 และไล่ได้ถึง z17
 *
 * กติกา fail-safe เดิม: ล้มเหลว → `{ok:false, reason}` ภาษาไทย ห้าม throw
 * ไทล์หายบางใบไม่ถือว่าล้มเหลว — ภาพโหว่เป็นหย่อมยังอ่านภูมิประเทศได้ ดีกว่าพื้นเรียบเปล่า
 *
 * `crossOrigin = 'anonymous'` **บังคับ** ต่างจาก `<img>` ธรรมดา: WebGL ปฏิเสธการอัป
 * texture จาก canvas ที่ปนเปื้อน ถ้าลืมจะได้ภูมิประเทศสีดำสนิทโดยไม่มี error ให้เห็น
 */
import { TILE, chooseZoom, tileRange, boundsAround, lonLatToPixel } from '../../services/dem-math.js';

export const ESRI_IMAGERY =
  'https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile';
const TIMEOUT_MS = 12_000;

/** Esri เรียง z/y/x ไม่ใช่ z/x/y — สลับแล้วได้ภาพคนละที่แบบไม่มี error */
export const esriTileUrl = (z, x, y) => `${ESRI_IMAGERY}/${z}/${y}/${x}`;

export function pickImageryZoom(lat, spanM) {
  return chooseZoom(lat, spanM, { maxTilesPerAxis: 6, zMin: 8, zMax: 17 });
}

export const browserDeps = {
  loadImage: url => new Promise((res, rej) => {
    const img = new Image();
    img.crossOrigin = 'anonymous';
    const t = setTimeout(() => rej(new Error('หมดเวลา')), TIMEOUT_MS);
    img.onload  = () => { clearTimeout(t); res({ width: img.naturalWidth, height: img.naturalHeight,
                                                 draw: (ctx, x, y) => ctx.drawImage(img, x, y) }); };
    img.onerror = () => { clearTimeout(t); rej(new Error('โหลดภาพไม่ได้')); };
    img.src = url;
  }),
  makeCanvas: (w, h) => { const c = document.createElement('canvas'); c.width = w; c.height = h; return c; },
};

/** ขอบเขตภูมิศาสตร์ของกรอบไทล์ (ไม่ใช่ของโดเมน) — ต้องใช้ map UV ให้ตรงตอนปู texture */
function mosaicBounds({ z, x0, x1, y0, y1 }) {
  const n = 2 ** z, deg = 360 / n;
  const latOf = ty => {
    const t = Math.PI * (1 - 2 * ty / n);
    return 180 / Math.PI * Math.atan(0.5 * (Math.exp(t) - Math.exp(-t)));
  };
  return { west: x0 * deg - 180, east: (x1 + 1) * deg - 180, north: latOf(y0), south: latOf(y1 + 1) };
}

export async function loadImageryMosaic(origin, spanM, deps = browserDeps) {
  try {
    const z = pickImageryZoom(origin.lat, spanM);
    const range = tileRange(boundsAround(origin, spanM), z);
    const { x0, x1, y0, y1 } = range;
    const cols = x1 - x0 + 1, rows = y1 - y0 + 1;
    const canvas = deps.makeCanvas(cols * TILE, rows * TILE);
    const ctx = canvas.getContext('2d');
    ctx.fillStyle = '#2a3442';                       // สีรองพื้นตรงไทล์ที่หาย
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    let got = 0;
    await Promise.all(Array.from({ length: cols * rows }, async (_, k) => {
      const cx = k % cols, cy = (k / cols) | 0;
      try {
        const img = await deps.loadImage(esriTileUrl(z, x0 + cx, y0 + cy));
        img.draw(ctx, cx * TILE, cy * TILE);
        got++;
      } catch { /* ไทล์เดียวหาย ไม่ล้มทั้งโมเสก */ }
    }));

    if (!got) return { ok: false, reason: 'โหลดภาพดาวเทียมไม่ได้เลยสักไทล์ — จะแสดงภูมิประเทศเป็นสีเรียบ' };
    return { ok: true, canvas, bounds: mosaicBounds(range), zoom: z, tiles: cols * rows, got };
  } catch (e) {
    return { ok: false, reason: 'โหลดภาพดาวเทียมไม่ได้ — ' + (e && e.message ? e.message : 'ไม่ทราบสาเหตุ') };
  }
}
```

```js
// app/src/map3d/three/volume-field.js
/**
 * volume-field — แปลงกริดความเข้มข้น**ระดับพื้น**ของเอนจินเป็นสนาม 3 มิติสำหรับ raymarch
 *
 * ไฟล์นี้คือจุดที่ความซื่อสัตย์ของโหมด 3 มิติทั้งหมดอยู่ อ่านให้จบก่อนแก้
 *
 * **รูปร่างแนวราบกับความเข้มข้นเป็นของจริง** มาจาก `grids[hour]` ที่เอนจินคำนวณ
 * รวมการเบนตามภูมิประเทศและสนามลมจริงถ้าเปิดไว้
 *
 * **ความหนาแนวดิ่งถูกสร้างขึ้น ไม่ใช่ผลการคำนวณ** เอนจินไม่มีสนาม 3 มิติให้
 *   field(i,j,k) = C₀(i,j) · vert(z_k) / vert(0)
 *   vert(z)      = exp(−(z−H)²/2σz²) + exp(−(z+H)²/2σz²)
 * เทอมหลังคือเงาสะท้อนที่พื้น ตัวเดียวกับใน `packages/engine/src/gaussian.ts`
 *
 * **ที่หาร vert(0) เพราะอยากให้ชั้น z=0 เท่ากับค่าที่เอนจินคำนวณเป๊ะ** ไม่ใช่ใกล้เคียง
 * นั่นคือสิ่งเดียวที่ทำให้พูดได้ว่า "ที่พื้นคือผลการคำนวณ เหนือพื้นคือการประมาณ"
 *
 * **ผลข้างเคียงที่ยอมรับแล้ว (ข้อจำกัดข้อ 6 ในสเปก)** ถ้าพลูมลอยสูงจนค่าที่พื้นเป็นศูนย์
 * ทั้งคอลัมน์จะเป็นศูนย์ คือ**มองไม่เห็นควันที่ลอยค้างอยู่ข้างบน** เกิดตอนอากาศเสถียรมาก
 * ตัวเลขในตารางยังถูก ภาพต่างหากที่บอกไม่ครบ ต้องมีป้ายบอกบนหน้าจอ
 * ทางเลือกคือคำนวณค่าบนอากาศจากสูตร Gaussian ใหม่ แต่จะทิ้งรูปร่างแนวราบที่เอนจิน
 * คำนวณจริงไป ซึ่งสำคัญกว่า จึงเลือกทางนี้
 *
 * `sigmas` กับ `downsampleMax` **import จาก volume.js ห้าม copy** ของเดิมเคยมีสูตร Briggs
 * สองชุดแล้วหลุดจากกันมาแล้วครั้งหนึ่ง (ดูคอมเมนต์หัวไฟล์ volume.js)
 */
import { sigmas, downsampleMax } from '../volume.js';

/** เพดานอัตราส่วน vert(z)/vert(0) — กันค่าระเบิดตอนพลูมลอยสูงจน vert(0) เข้าใกล้ศูนย์ */
export const RATIO_MAX = 1e4;

export function buildField({ grid, res, hour, step = 4, Nz = 48, pexag = 1 }) {
  const { N, cell, cx, cy, R } = res;
  const { data: flatGrid, M: Nx } = downsampleMax(grid, N, step);
  const Ny = Nx, bcell = cell * step;

  // ความสูงพลูม: เฟสคุกรุ่นเป็นตัวกำหนดค่าที่พื้น ถ้าไม่มีจึงใช้เฟสเปลวไฟ (เหมือน volume.js)
  const H0 = (hour.qSm > 0 ? hour.Hsm : hour.Hfl) || 0;
  const lid0 = Math.max(hour.mix, 60);

  // pexag ยืดแกนดิ่งทั้งแกนอย่างสม่ำเสมอ ค่าที่ z=0 จึงไม่ขยับเพราะ vert(0)/vert(0) = 1 เสมอ
  const H = H0 * pexag, lid = lid0 * pexag;

  // σz เฉลี่ยไว้ตัดสิน zTop เท่านั้น ตัวจริงคำนวณรายคอลัมน์ข้างล่าง
  const szMid = Math.min(sigmas(Math.max(R / 2, 12), hour.stab)[1], lid0 / 1.25) * pexag;
  const zTop = Math.min(lid, Math.max(H + 3 * szMid, 60 * pexag));

  const out = new Float32Array(Nx * Ny * Nz);
  let vmax = 0;

  for (let bj = 0; bj < Ny; bj++) {
    for (let bi = 0; bi < Nx; bi++) {
      const c0 = flatGrid[bj * Nx + bi];
      if (!(c0 > 0)) continue;                 // คอลัมน์ว่างข้ามไป (ดูข้อจำกัดข้อ 6 ข้างบน)

      // ศูนย์กลางบล็อกในพิกัดเอนจิน — กองไฟอยู่ที่ (0,0) ศูนย์กลางกริดเลื่อนไปท้ายลมที่ (cx,cy)
      const xm = cx - R + (bi + 0.5) * bcell;
      const ym = cy + R - (bj + 0.5) * bcell;
      const d = Math.hypot(xm, ym);

      const sz = Math.min(sigmas(Math.max(d, 12), hour.stab)[1], lid0 / 1.25) * pexag;
      const inv2s2 = 1 / (2 * sz * sz);
      const v0 = Math.exp(-H * H * inv2s2) * 2;          // vert(0): สองเทอมเท่ากันพอดีที่ z=0
      const invV0 = v0 > 0 ? 1 / v0 : 0;

      for (let k = 0; k < Nz; k++) {
        const z = (k + 0.5) / Nz * zTop;
        const a = z - H, b = z + H;
        const vert = Math.exp(-a * a * inv2s2) + Math.exp(-b * b * inv2s2);
        const ratio = Math.min(vert * invV0, RATIO_MAX);
        const v = c0 * ratio;
        out[k * Nx * Ny + bj * Nx + bi] = v;
        if (v > vmax) vmax = v;
      }
    }
  }
  return { data: out, Nx, Ny, Nz, zTop, vmax };
}

/* ---------------- LUT สี ---------------- */
/**
 * สเกลของ LUT เป็นล็อก เพราะแถบ AQI ห่างกันแบบทวีคูณ (15 · 25 · 37.5 · 75 · 150 · 350)
 * สเกลเชิงเส้นจะยัดสี่แถบล่างไว้ในสิบเปอร์เซ็นต์แรกของ texture จนแยกไม่ออก
 */
export function lutCoord(c, { lo = 1, hi = 500 } = {}) {
  if (!(c > lo)) return 0;
  const t = Math.log(c / lo) / Math.log(hi / lo);
  return t >= 1 ? 1 : t;
}

const hex = h => [parseInt(h.slice(1, 3), 16), parseInt(h.slice(3, 5), 16), parseInt(h.slice(5, 7), 16)];

/** สร้าง RGBA ไบต์ของ LUT จาก BANDS ชุดเดียวกับแผนที่ 2D — ห้ามคิดสีใหม่ที่นี่ */
export function buildLutBytes(bands, { lo = 1, hi = 500, size = 256 } = {}) {
  const out = new Uint8Array(size * 4);
  const below = [0x9f, 0xb0, 0xc4];                       // ต่ำกว่าแถบแรก: เทาอมฟ้า เหมือน volume.js
  for (let i = 0; i < size; i++) {
    const c = lo * Math.pow(hi / lo, i / (size - 1));
    let rgb = below;
    for (let b = bands.length - 1; b >= 0; b--) if (c >= bands[b].lo) { rgb = hex(bands[b].c); break; }
    out[i * 4] = rgb[0]; out[i * 4 + 1] = rgb[1]; out[i * 4 + 2] = rgb[2]; out[i * 4 + 3] = 255;
  }
  return out;
}
```

```js
// app/src/map3d/three/terrain.js
/**
 * terrain — เมชภูมิประเทศ และ texture ความสูงที่ shader ของควันใช้ตัดรังสี
 *
 * เงาเขาที่นี่**มาจากแสงจริงส่องเมชที่มี normal** ไม่ใช่ hillshade ที่คำนวณทับภาพ
 * ต่างจากฝั่ง MapLibre ที่ล็อกมุมแสง 315° ตายตัว — ที่นี่ทิศเงาเปลี่ยนตามเวลาในวันจริง
 *
 * จุดยอดอยู่ที่**ศูนย์กลางเซลล์** เมชจึงกว้าง `2R − cell` ไม่ใช่ `2R`
 * เลื่อนครึ่งเซลล์แล้วเงากับภาพดาวเทียมจะเหลื่อมกันแบบที่ตาจับได้
 *
 * `elevTexture` ปูเต็มโดเมน `[cx−R, cx+R]` โดยเท็กเซล `i` อยู่ที่ `uv=(i+0.5)/N`
 * ซึ่งตรงกับศูนย์กลางเซลล์พอดี — shader จึงสุ่มความสูงได้ตรงโดยไม่ต้องชดเชยอะไร
 */
import * as THREE from 'three';

export function buildTerrain({ elev, res, exag = 1, texture = null }) {
  const { N, cell, cx, cy, R } = res;
  const span = 2 * R - cell;
  const geo = new THREE.PlaneGeometry(span, span, N - 1, N - 1);
  const pos = geo.attributes.position;

  const raw = elev && elev.length === N * N ? elev : new Float32Array(N * N);
  for (let j = 0; j < N; j++)
    for (let i = 0; i < N; i++)
      pos.setZ(j * N + i, raw[j * N + i] * exag);
  pos.needsUpdate = true;
  geo.computeVertexNormals();

  const mat = new THREE.MeshStandardMaterial({
    map: texture, color: texture ? 0xffffff : 0x5d6b52, roughness: 0.95, metalness: 0,
  });
  const mesh = new THREE.Mesh(geo, mat);
  mesh.position.set(cx, cy, 0);
  mesh.renderOrder = 0;

  // ความสูง "จริง" (ยังไม่คูณ exag) สำหรับ shader — ตัวคูณส่งแยกเป็น uniform
  const elevTexture = new THREE.DataTexture(raw, N, N, THREE.RedFormat, THREE.FloatType);
  elevTexture.minFilter = elevTexture.magFilter = THREE.LinearFilter;
  elevTexture.wrapS = elevTexture.wrapT = THREE.ClampToEdgeWrapping;
  elevTexture.needsUpdate = true;

  return {
    mesh, elevTexture, raw,
    setExaggeration(v) {
      for (let k = 0; k < N * N; k++) pos.setZ(k, raw[k] * v);
      pos.needsUpdate = true;
      geo.computeVertexNormals();
    },
    setTexture(tex) { mat.map = tex; mat.color.set(tex ? 0xffffff : 0x5d6b52); mat.needsUpdate = true; },
    dispose() { geo.dispose(); mat.dispose(); elevTexture.dispose(); },
  };
}
```

```js
// app/src/map3d/three/scene.js
/**
 * scene — renderer กล้อง controls ฟ้า หมอก แสง และห่วงวาด
 *
 * **แกน Z ชี้ขึ้น** ตั้ง `camera.up` เอง เพราะค่าปริยายของ three.js คือ Y ขึ้น
 * เลือก Z ขึ้นเพราะทำให้ `x, y` ในฉากเป็นพิกัดเอนจินตรงๆ ไม่ต้องแปลงที่ไหนเลย
 * และ `PlaneGeometry`/`BoxGeometry` วางตัวบนระนาบ XY อยู่แล้ว จึงไม่ต้องหมุนอะไร
 *
 * กล้องเริ่มที่มุมต่ำและใกล้พื้น ด้วยเหตุผลเดียวกับฝั่ง MapLibre (`pitch: 74`)
 * มุมสูงกับซูมออกทำให้ภูเขาดูแบนจนอ่านความชันไม่ออก
 *
 * ห่วงวาดต้อง `stop()` ตอนสลับกลับ 2D ไม่งั้นกิน GPU ค้างทั้งที่ไม่มีใครดู
 */
import * as THREE from 'three';
import { OrbitControls } from 'three/examples/jsm/controls/OrbitControls.js';
import { skyFor, sunDirection, isDaylight } from '../sky-palette.js';

export function createScene(container, { origin, hourKey, spanM }) {
  const renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'high-performance' });
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));   // เกิน 2 ไม่ต่างที่ตา แต่ raymarch ช้าเป็นเท่า
  renderer.setSize(container.clientWidth || 1, container.clientHeight || 1);
  container.appendChild(renderer.domElement);

  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(55, 1, 10, spanM * 8);
  camera.up.set(0, 0, 1);
  camera.position.set(0, -spanM * 0.9, spanM * 0.28);      // ใต้ลมมองขึ้นเหนือ มุมต่ำ

  const controls = new OrbitControls(camera, renderer.domElement);
  controls.enableDamping = true;
  controls.maxPolarAngle = Math.PI * 0.495;                // กันกล้องมุดใต้พื้น
  controls.target.set(0, 0, 0);

  const sun = new THREE.DirectionalLight(0xfff4e0, 2.2);
  const ambient = new THREE.AmbientLight(0x8fa6c4, 0.9);
  scene.add(sun, ambient);

  function setHour(key) {
    const p = skyFor(key);
    scene.background = new THREE.Color(p.sky);
    scene.fog = new THREE.FogExp2(new THREE.Color(p.fog), 1.1 / spanM);
    const d = sunDirection(key, origin.lat);
    // ดวงอาทิตย์ใต้ขอบฟ้า → ยกขึ้นเล็กน้อยและหรี่ลง ไม่งั้นภูเขาดำสนิทจนอ่านไม่ออก
    const up = isDaylight(key) ? d.z : 0.25;
    sun.position.set(d.x * spanM, d.y * spanM, Math.max(up, 0.12) * spanM);
    sun.intensity = isDaylight(key) ? 2.2 : 0.45;
    ambient.intensity = isDaylight(key) ? 0.9 : 0.55;
  }
  setHour(hourKey);

  let raf = 0, running = false;
  function frame() { if (!running) return; controls.update(); renderer.render(scene, camera); raf = requestAnimationFrame(frame); }

  return {
    scene, camera, renderer, controls,
    add: o => scene.add(o),
    remove: o => scene.remove(o),
    setHour,
    resize() {
      const w = container.clientWidth || 1, h = container.clientHeight || 1;
      renderer.setSize(w, h); camera.aspect = w / h; camera.updateProjectionMatrix();
    },
    start() { if (!running) { running = true; frame(); } },
    stop()  { running = false; cancelAnimationFrame(raf); },
    dispose() {
      running = false; cancelAnimationFrame(raf);
      controls.dispose(); renderer.dispose();
      if (renderer.domElement.parentNode) renderer.domElement.parentNode.removeChild(renderer.domElement);
    },
  };
}
```

```js
// app/src/map3d/three/index.js
/**
 * index — facade เดียวที่ app.js คุยด้วย
 *
 * ตั้งใจให้ app.js ไม่รู้จัก three.js เลยแม้แต่ import เดียว จะได้ถอดหรือเปลี่ยน
 * ตัววาดทีหลังโดยไม่ต้องรื้อไฟล์ 1900 บรรทัด
 *
 * `view` เป็นก้อนข้อมูลธรรมดา ไม่ใช่ `S` ทั้งก้อน — ทดสอบง่ายและอ่านออกว่าพึ่งอะไรบ้าง
 */
import { createScene } from './scene.js';
import { buildTerrain } from './terrain.js';
import { loadImageryMosaic } from './basemap.js';

export async function create3D(container, view) {
  const THREE = await import('three');
  const { origin, result } = view;
  const spanM = result ? 2 * result.R : 20000;
  const stage = createScene(container, { origin, hourKey: view.hourKey, spanM });

  let terrain = null, notes = [];

  if (result) {
    terrain = buildTerrain({ elev: view.elev, res: result, exag: view.opts.exag });
    stage.add(terrain.mesh);
    stage.controls.target.set(result.cx, result.cy, 0);

    const mos = await loadImageryMosaic(origin, result.R * 1.4);
    if (mos.ok) {
      const tex = new THREE.CanvasTexture(mos.canvas);
      tex.colorSpace = THREE.SRGBColorSpace;
      // ตัด UV ให้ตรงโดเมนจริง — โมเสกครอบกว้างกว่าโดเมนเสมอเพราะขอบไทล์ไม่ลงตัว
      applyDomainUv(terrain.mesh.geometry, result, origin, mos.bounds);
      terrain.setTexture(tex);
    } else {
      notes.push(mos.reason);
    }
    if (!view.elev) notes.push('ไม่มีข้อมูลความสูง — แสดงเป็นพื้นราบ');
  }

  stage.start();

  return {
    notes,
    update(next) {
      stage.setHour(next.hourKey);
      if (terrain && next.opts.exag !== view.opts.exag) terrain.setExaggeration(next.opts.exag);
      view = next;
    },
    resize: () => stage.resize(),
    fitBounds() {
      const r = view.result;
      if (!r) return;
      stage.controls.target.set(r.cx, r.cy, 0);
      stage.camera.position.set(r.cx, r.cy - r.R * 1.8, r.R * 0.6);
    },
    dispose() { stage.stop(); if (terrain) terrain.dispose(); stage.dispose(); },
  };
}

/** map UV ของเมชให้ชี้ไปยังส่วนของโมเสกที่ตรงกับโดเมนจริง */
function applyDomainUv(geo, res, origin, bounds) {
  const { N, cell, cx, cy, R } = res;
  const M_LAT = 111320, mLon = 111320 * Math.cos(origin.lat * Math.PI / 180);
  const uv = geo.attributes.uv;
  for (let j = 0; j < N; j++) {
    for (let i = 0; i < N; i++) {
      const x = cx - R + (i + 0.5) * cell, y = cy + R - (j + 0.5) * cell;
      const lng = origin.lng + x / mLon, lat = origin.lat + y / M_LAT;
      uv.setXY(j * N + i,
        (lng - bounds.west) / (bounds.east - bounds.west),
        (lat - bounds.south) / (bounds.north - bounds.south));
    }
  }
  uv.needsUpdate = true;
}
```

## งาน 5 — ก้อนควัน raymarch (เป้าหมายหลักของทั้งแผน)

จบงานนี้ **เห็นก้อนควันฟูๆ และภูเขาบังควันได้จริง** ซึ่งเป็นสิ่งที่ `fill-extrusion` ทำไม่ได้

**ไฟล์:**
- สร้าง: `app/src/map3d/three/volume.glsl.js` · `volume-mesh.js`
- แก้: `app/src/map3d/three/index.js` — ต่อก้อนควันเข้าฉาก

**รอยต่อ:**
- ใช้จากงานก่อน: `buildField` `buildLutBytes` `RATIO_MAX` (งาน 3) · `terrain.elevTexture` (งาน 4)
- ส่งต่อ: `createVolume({res, elevTexture, bands}) → {mesh, update({grid, hour, opts, bg}), dispose()}`

**สิ่งที่พลาดง่ายและต้องระวัง:**
- **กล่องวางที่ `(cx, cy)` ไม่ใช่ที่กองไฟ** ศูนย์กลางกริดเลื่อนไปท้ายลม วางผิดควันเหลื่อมทั้งก้อน
- **ต้อง jitter จุดเริ่มรังสี** ไม่งั้นเห็นเป็นวงแหวนซ้อนชัดมาก
- `depthWrite: false` แต่ `depthTest: true` — ให้ภูเขาที่อยู่**หน้า**กล่องบังได้ด้วย
  ส่วนภูเขาที่อยู่**ใน**กล่องใช้การอ่าน `uElev` ตัดรังสี
- `side: THREE.BackSide` เพื่อให้ยังเห็นควันตอนกล้องเข้าไปอยู่ในกล่อง

```js
// app/src/map3d/three/volume.glsl.js
/**
 * shader ของก้อนควัน — เดินรังสีผ่านสนาม 3 มิติแล้วสะสมสีจากหน้าไปหลัง
 *
 * **ทำไมต้อง raymarch ไม่ใช่กล่องทึบ** กล่องของ fill-extrusion บังภูเขาเสมอเพราะเป็นทรงตัน
 * ทั้งที่คำถามหลักของทั้งโปรเจกต์คือ "ภูเขาเบนลมหรือลมข้ามไป" การเดินรังสีทำให้
 * หยุดที่ผิวดินได้จริง และค่าที่เห็นคือผลรวมตามทางเดินของแสง ซึ่งตรงกับความหมายของควัน
 *
 * **ตัดด้วยผิวดินโดยตรง ไม่ทำ depth prepass** เรามี DEM เป็น texture อยู่แล้ว
 * และภูมิประเทศเป็น heightfield ไม่มีอะไรยื่นเกิน การเทียบ `z < elev(x,y)` จึงถูกต้องพอดี
 * และประหยัดกว่าการตั้ง render target พร้อม depth texture ทั้งชุด
 */

export const volumeVert = /* glsl */`
varying vec3 vLocal;
void main() {
  vLocal = position;                       // พิกัดในระบบของกล่อง จุดกำเนิดอยู่กลางกล่อง
  gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
}`;

export const volumeFrag = /* glsl */`
precision highp float;
precision highp sampler3D;

varying vec3 vLocal;

uniform vec3  uBoxSize;      // (2R, 2R, zTop)
uniform vec3  uCamLocal;     // ตำแหน่งกล้องในระบบพิกัดกล่อง
uniform sampler3D uField;
uniform sampler2D uElev;
uniform sampler2D uLut;
uniform float uVmax;         // ค่าสูงสุดของสนาม ใช้ถอดค่ากลับจาก texture ที่ normalize แล้ว
uniform float uK;            // สัมประสิทธิ์ความทึบ ผูกกับสไลเดอร์ความทึบควัน
uniform float uElevScale;    // ตัวคูณยกภูมิประเทศ ต้องตรงกับที่เมชใช้
uniform float uLutLo;
uniform float uLutHi;
uniform int   uSteps;

/* สุ่มค่าจากพิกัดจอ — จุดเริ่มรังสีที่ไม่สุ่มจะทำให้เห็นวงแหวนซ้อนเป็นชั้นชัดมาก */
float hash(vec2 p) { return fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453); }

/* ตัดกล่องด้วยวิธี slab คืนช่วง t ที่รังสีอยู่ในกล่อง */
vec2 boxRange(vec3 o, vec3 d, vec3 half_) {
  vec3 inv = 1.0 / d;
  vec3 a = (-half_ - o) * inv, b = (half_ - o) * inv;
  vec3 lo = min(a, b), hi = max(a, b);
  return vec2(max(max(lo.x, lo.y), lo.z), min(min(hi.x, hi.y), hi.z));
}

void main() {
  vec3 half_ = uBoxSize * 0.5;
  vec3 ro = uCamLocal;
  vec3 rd = normalize(vLocal - uCamLocal);

  vec2 tr = boxRange(ro, rd, half_);
  float t0 = max(tr.x, 0.0), t1 = tr.y;
  if (t1 <= t0) discard;

  float steps = float(uSteps);
  float dt = (t1 - t0) / steps;
  float t = t0 + dt * hash(gl_FragCoord.xy);       // jitter กันแถบวงแหวน

  vec4 acc = vec4(0.0);
  for (int i = 0; i < 256; i++) {
    if (float(i) >= steps || acc.a > 0.98) break;

    vec3 p = ro + rd * t;
    vec3 uvw = (p + half_) / uBoxSize;

    // จมใต้ผิวดินแล้วหยุด — นี่คือจุดที่ภูเขาบังควันได้จริง
    float ground = texture(uElev, uvw.xy).r * uElevScale;
    float worldZ = p.z + half_.z;
    if (worldZ < ground) break;

    float c = texture(uField, uvw).r * uVmax;
    if (c > uLutLo) {
      float u = clamp(log(c / uLutLo) / log(uLutHi / uLutLo), 0.0, 1.0);
      vec3 col = texture(uLut, vec2(u, 0.5)).rgb;
      float a = 1.0 - exp(-uK * c * dt);          // Beer–Lambert
      acc.rgb += (1.0 - acc.a) * col * a;
      acc.a   += (1.0 - acc.a) * a;
    }
    t += dt;
  }

  if (acc.a <= 0.002) discard;
  gl_FragColor = vec4(acc.rgb / max(acc.a, 1e-4), acc.a);
}`;
```

- [ ] **ขั้น 2: เขียน `volume-mesh.js`**

```js
// app/src/map3d/three/volume-mesh.js
/**
 * volume-mesh — ประกอบสนาม 3 มิติ + LUT + shader เข้าเป็นก้อนควันหนึ่งก้อน
 *
 * **กล่องวางที่ `(cx, cy)` ไม่ใช่ที่กองไฟ** ศูนย์กลางกริดของเอนจินเลื่อนไปทางท้ายลม
 * (ดู GridSpec ใน packages/engine/src/types.ts) วางที่กองไฟแล้วควันจะเหลื่อมทั้งก้อน
 * แบบที่ดูเผินๆ เหมือนถูก
 *
 * สนามถูก normalize ด้วย `vmax` ก่อนอัปเป็น half-float แล้วคูณกลับใน shader
 * half-float เก็บค่าถึง 65504 ก็จริง แต่ความละเอียดที่ปลายบนหยาบ การ normalize
 * ให้อยู่ใน [0,1] ใช้ช่วงที่แม่นที่สุดของรูปแบบนี้
 *
 * `depthWrite:false` แต่ `depthTest:true` — ภูเขาที่อยู่หน้ากล่องบังด้วย depth buffer
 * ส่วนภูเขาที่อยู่ในกล่องบังด้วยการอ่าน uElev ใน shader ต้องมีทั้งสองทาง
 */
import * as THREE from 'three';
import { buildField, buildLutBytes } from './volume-field.js';
import { volumeVert, volumeFrag } from './volume.glsl.js';

const LUT_LO = 1, LUT_HI = 500;

export function createVolume({ res, elevTexture, bands }) {
  const { cx, cy, R } = res;

  const lut = new THREE.DataTexture(buildLutBytes(bands, { lo: LUT_LO, hi: LUT_HI, size: 256 }),
                                    256, 1, THREE.RGBAFormat);
  lut.minFilter = lut.magFilter = THREE.LinearFilter;
  lut.wrapS = lut.wrapT = THREE.ClampToEdgeWrapping;
  lut.needsUpdate = true;

  const uniforms = {
    uBoxSize:   { value: new THREE.Vector3(2 * R, 2 * R, 1) },
    uCamLocal:  { value: new THREE.Vector3() },
    uField:     { value: null },
    uElev:      { value: elevTexture },
    uLut:       { value: lut },
    uVmax:      { value: 1 },
    uK:         { value: 0.02 },
    uElevScale: { value: 1 },
    uLutLo:     { value: LUT_LO },
    uLutHi:     { value: LUT_HI },
    uSteps:     { value: 96 },
  };

  const mat = new THREE.ShaderMaterial({
    vertexShader: volumeVert, fragmentShader: volumeFrag, uniforms,
    transparent: true, depthWrite: false, depthTest: true, side: THREE.BackSide,
    glslVersion: THREE.GLSL3,
  });

  const geo = new THREE.BoxGeometry(1, 1, 1);
  const mesh = new THREE.Mesh(geo, mat);
  mesh.renderOrder = 10;
  mesh.frustumCulled = false;              // กล้องอยู่ในกล่องได้ ปล่อยให้ cull จะกะพริบหาย
  mesh.visible = false;

  let field = null;

  return {
    mesh,
    update({ grid, hour, opts, step }) {
      if (!grid || !hour) { mesh.visible = false; return; }
      field = buildField({ grid, res, hour, step, Nz: 48, pexag: opts.pexag });
      if (!(field.vmax > 0)) { mesh.visible = false; return; }

      const { data, Nx, Ny, Nz, zTop, vmax } = field;
      const norm = new Uint16Array(data.length);
      for (let i = 0; i < data.length; i++) norm[i] = toHalf(data[i] / vmax);

      if (uniforms.uField.value) uniforms.uField.value.dispose();
      const tex = new THREE.Data3DTexture(norm, Nx, Ny, Nz);
      tex.format = THREE.RedFormat; tex.type = THREE.HalfFloatType;
      tex.minFilter = tex.magFilter = THREE.LinearFilter;
      tex.wrapS = tex.wrapT = tex.wrapR = THREE.ClampToEdgeWrapping;
      tex.needsUpdate = true;
      uniforms.uField.value = tex;

      uniforms.uVmax.value = vmax;
      uniforms.uBoxSize.value.set(2 * R, 2 * R, zTop);
      uniforms.uElevScale.value = opts.exag;
      // ความทึบ: สไลเดอร์ 0.1–1 แมปเป็นสัมประสิทธิ์ที่ให้ก้อนกลางๆ อ่านออกพอดี
      uniforms.uK.value = 0.004 + opts.smokeOpacity * 0.05;

      geo.dispose();
      mesh.geometry = new THREE.BoxGeometry(2 * R, 2 * R, zTop);
      mesh.position.set(cx, cy, zTop / 2);
      mesh.visible = true;
    },
    /** ต้องเรียกทุกเฟรม — shader ทำงานในระบบพิกัดกล่อง จึงต้องรู้กล้องในระบบนั้น */
    syncCamera(camera) {
      uniforms.uCamLocal.value.copy(camera.position);
      mesh.worldToLocal(uniforms.uCamLocal.value);
    },
    dispose() {
      if (uniforms.uField.value) uniforms.uField.value.dispose();
      lut.dispose(); mat.dispose(); mesh.geometry.dispose();
    },
  };
}

/* แปลง float32 เป็นบิตของ half-float — three ไม่ได้ export ตัวช่วยนี้ออกมาให้ */
function toHalf(v) {
  const f = new Float32Array(1), i = new Int32Array(f.buffer);
  f[0] = v;
  const x = i[0], sign = (x >> 16) & 0x8000;
  let exp = ((x >> 23) & 0xff) - 112, man = x & 0x7fffff;
  if (exp <= 0) return sign;
  if (exp >= 31) return sign | 0x7bff;
  return sign | (exp << 10) | (man >> 13);
}
```

```js
// app/src/map3d/three/drape.js
/**
 * drape — ปูรูปหลายเหลี่ยมและภาพให้เลาะไปตามผิวภูมิประเทศ
 *
 * ถ้าวางแปลงเผาไว้ที่ z=0 เฉยๆ มันจะจมหายใต้ภูเขาทันทีที่พื้นตรงนั้นสูงกว่าศูนย์
 * ทุกจุดจึงต้องยกตามความสูงจริงของพื้นใต้จุดนั้น แล้วยกเพิ่มอีกนิด (`lift`)
 * กัน z-fighting กับเมชภูมิประเทศ
 */

/** ความสูงพื้นที่พิกัดเอนจิน (x,y) — bilinear บนกริดที่ค่าอยู่ตรงศูนย์กลางเซลล์ */
export function elevAt(raw, res, x, y) {
  if (!raw) return 0;
  const { N, cell, cx, cy, R } = res;
  const fi = (x - (cx - R)) / cell - 0.5;
  const fj = ((cy + R) - y) / cell - 0.5;
  const cl = (v, hi) => (v < 0 ? 0 : v > hi ? hi : v);
  const i0 = cl(Math.floor(fi), N - 1), j0 = cl(Math.floor(fj), N - 1);
  const i1 = cl(i0 + 1, N - 1), j1 = cl(j0 + 1, N - 1);
  const tx = cl(fi - i0, 1), ty = cl(fj - j0, 1);
  const a = raw[j0 * N + i0], b = raw[j0 * N + i1], c = raw[j1 * N + i0], d = raw[j1 * N + i1];
  return (a * (1 - tx) + b * tx) * (1 - ty) + (c * (1 - tx) + d * tx) * ty;
}

/** วงรอบในพิกัดเอนจิน → จุดยอด 3 มิติที่เลาะผิวดิน */
export function drapeRing(points, { raw, res, exag = 1, lift = 6 }) {
  const out = new Float32Array(points.length * 3);
  for (let k = 0; k < points.length; k++) {
    const [x, y] = points[k];
    out[k * 3] = x; out[k * 3 + 1] = y;
    out[k * 3 + 2] = elevAt(raw, res, x, y) * exag + lift;
  }
  return out;
}
```

```js
// app/src/map3d/three/markers.js
/**
 * markers — แปลงเผาและตัวรับในฉาก 3 มิติ
 *
 * ทั้งคู่ปูเลาะผิวดินผ่าน `drape.js` เพราะถ้าวางที่ z=0 จะจมใต้ภูเขา
 * สีของตัวรับต้องมาจากตัวคำนวณสีเดียวกับฝั่ง 2D จึงรับ `colorOf` เข้ามา ไม่คิดสีเอง
 */
import * as THREE from 'three';
import { elevAt, drapeRing } from './drape.js';

export function createMarkers({ res, raw, exag = 1 }) {
  const group = new THREE.Group();
  let cur = { raw, exag };
  const own = [];

  function clear() { for (const o of own) { group.remove(o); o.geometry.dispose(); o.material.dispose(); } own.length = 0; }

  return {
    group,
    setTerrain(nextRaw, nextExag) { cur = { raw: nextRaw, exag: nextExag }; },
    update({ plots, receptors, ringsOf, colorOf }) {
      clear();
      for (const p of plots) {
        const ring = ringsOf(p);
        if (!ring || ring.length < 3) continue;
        const verts = drapeRing(ring, { raw: cur.raw, res, exag: cur.exag, lift: 8 });
        const g = new THREE.BufferGeometry();
        g.setAttribute('position', new THREE.BufferAttribute(verts, 3));
        const line = new THREE.LineLoop(g, new THREE.LineBasicMaterial({ color: 0xff8a6a }));
        group.add(line); own.push(line);
      }
      for (let i = 0; i < receptors.length; i++) {
        const r = receptors[i];
        const g = new THREE.SphereGeometry(Math.max(res.cell * 0.6, 40), 10, 8);
        const m = new THREE.MeshBasicMaterial({ color: new THREE.Color(colorOf(i)) });
        const s = new THREE.Mesh(g, m);
        s.position.set(r.x, r.y, elevAt(cur.raw, res, r.x, r.y) * cur.exag + 30);
        group.add(s); own.push(s);
      }
    },
    dispose() { clear(); },
  };
}

/** ภาพชั้นควัน 2 มิติทาบลงบนพื้น — ระนาบย่อยละเอียดพอจะเลาะภูเขาได้ */
export function createGroundImage({ res }) {
  const { N, cell, cx, cy, R } = res;
  const seg = Math.min(N - 1, 128);
  const geo = new THREE.PlaneGeometry(2 * R - cell, 2 * R - cell, seg, seg);
  const mat = new THREE.MeshBasicMaterial({ transparent: true, depthWrite: false });
  const mesh = new THREE.Mesh(geo, mat);
  mesh.position.set(cx, cy, 0);
  mesh.renderOrder = 1;
  mesh.visible = false;
  let tex = null;

  return {
    mesh,
    update({ raster, opacity, visible, raw, exag }) {
      mesh.visible = !!(visible && raster);
      if (!mesh.visible) return;
      mat.opacity = opacity;
      if (tex && tex.image && tex.image.src === raster.url) { /* ภาพเดิม ไม่ต้องโหลดซ้ำ */ }
      else {
        if (tex) tex.dispose();
        tex = new THREE.TextureLoader().load(raster.url);
        tex.colorSpace = THREE.SRGBColorSpace;
        mat.map = tex; mat.needsUpdate = true;
      }
      const pos = geo.attributes.position;
      for (let k = 0; k < pos.count; k++)
        pos.setZ(k, elevAt(raw, res, cx + pos.getX(k), cy + pos.getY(k)) * exag + 3);
      pos.needsUpdate = true;
    },
    dispose() { if (tex) tex.dispose(); geo.dispose(); mat.dispose(); },
  };
}
```
